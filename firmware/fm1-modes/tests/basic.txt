# The modes firmware in fm1-base's host simulator: tools/build.py ../fm1-modes --host (from fm1-base), then
#   ../fm1-modes/build/host/modes_host ../fm1-modes/tests/basic.txt OUT_DIR
# The first time, with no key held, the first mode starts: Almanac.
wait 600
expect chooser 0
expect mode 0
expect month 0
expect battery 3
shot almanac.png
# January plays its chord, open fifths, on each key.
key 7 down
wait 300
expect notes 3
key 7 up
wait 30
# April's chord as a pattern of raindrops.
turn PRESET 3
wait 50
expect month 3
expect play 3
key 7 down
wait 1200
expect arp 1
expect arp_notes 4
key 7 up
wait 30
expect arp 0
turn ALGO 1
wait 50
expect sky 3
wait 5000
expect dirty 0
# A piano key held while switching on brings up the chooser; G picks Woof.
key 0 down
reboot
wait 600
expect chooser 1
shot chooser.png
key 0 up
wait 50
tapkey 2
wait 50
expect chooser 0
expect mode 1
expect dog 0
# The key that chose does not bark.
expect notes 0
wait 1800
key 7 down
wait 120
expect notes 1
expect voices 1
shot woof.png
key 7 up
wait 30
turn PRESET 1
wait 50
expect dog 1
expect play 1
key 7 down
wait 1200
expect arp 1
key 7 up
wait 30
expect arp 0
press GLO
release GLO
wait 30
expect show 1
wait 10300
expect dog 2
press GLO
release GLO
wait 2000
expect dirty 0
# Switched on again with no key: Woof again, as it was left.
reboot
wait 600
expect chooser 0
expect mode 1
expect dog 2
expect show 0
# And back to Almanac, which kept April under a foggy sky.
key 5 down
reboot
wait 600
expect chooser 1
key 5 up
wait 50
tapkey 0
wait 50
expect mode 0
expect month 3
expect sky 3
wait 5000
expect dirty 0
reboot
wait 600
expect mode 0
